# mcoder M0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the foundation of mcoder — a Rust server + Node TUI that runs a coding agent capable of reading, editing, and running code in a local project.

**Architecture:** Modular single-binary Rust server (tokio + axum/tungstenite + sqlx + tree-sitter) communicating over WebSocket + JSON-RPC with a Node Ink TUI client. Embedded mode: `mcoder` spawns both server and TUI.

**Tech Stack:** Rust 1.80+, tokio, serde, sqlx (SQLite + FTS5), tree-sitter, tokio-tungstenite, thiserror, anyhow, clap, sha2, yargs-parser (Node), Ink (Node), zustand (Node), ws (Node)

use crate::types::AppConfig;
use anyhow::{Context, Result};
use dirs::home_dir;
use std::path::{Path, PathBuf};

pub fn global_config_dir() -> PathBuf {
    home_dir()
        .map(|h| h.join(".mcoder"))
        .unwrap_or_else(|| PathBuf::from(".mcoder"))
}

pub fn project_config_dir(project: &Path) -> PathBuf {
    project.join(".mcoder")
}

pub fn load_config(project: Option<&Path>) -> Result<AppConfig> {
    let global_path = global_config_dir().join("config.toml");
    let global = load_toml::<AppConfig>(&global_path).unwrap_or_default();

    if let Some(proj) = project {
        let proj_path = project_config_dir(proj).join("config.toml");
        if let Some(proj_config) = load_toml::<AppConfig>(&proj_path) {
            return Ok(merge_configs(global, proj_config));
        }
    }

    Ok(global)
}

fn load_toml<T: serde::de::DeserializeOwned>(path: &Path) -> Option<T> {
    if !path.exists() {
        return None;
    }
    let content = std::fs::read_to_string(path).ok()?;
    toml::from_str(&content).ok()
}

fn merge_configs(mut base: AppConfig, overlay: AppConfig) -> AppConfig {
    if !overlay.default_model.is_empty() {
        base.default_model = overlay.default_model;
    }
    base.models.extend(overlay.models);
    base.roles.extend(overlay.roles);
    if overlay.loop_max_iters != 0 {
        base.loop_max_iters = overlay.loop_max_iters;
    }
    if overlay.compact.strategy != "auto" {
        base.compact = overlay.compact;
    }
    if overlay.tui.compact != base.tui.compact {
        base.tui = overlay.tui;
    }
    if overlay.server.host != "127.0.0.1" || overlay.server.port != 7654 {
        base.server = overlay.server;
    }
    base
}

impl Default for AppConfig {
    fn default() -> Self {
        Self {
            default_model: "gpt-4o".into(),
            models: std::collections::HashMap::new(),
            roles: std::collections::HashMap::new(),

                threshold: 0.8,

pub mod jsonl;
pub mod sandbox;

use anyhow::{Context, Result};
use sqlx::{sqlite::SqlitePoolOptions, SqlitePool};
use std::path::Path;

pub type DbPool = SqlitePool;

pub async fn init_sqlite(db_path: &Path) -> Result<DbPool> {
    if let Some(parent) = db_path.parent() {
        std::fs::create_dir_all(parent)?;
    }

    let pool = SqlitePoolOptions::new()
        .max_connections(5)
        .connect_with(
            sqlx::sqlite::SqliteConnectOptions::new()
                .filename(db_path)
                .create_if_missing(true)
                .busy_timeout(std::time::Duration::from_secs(5)),
        )
        .await
        .with_context(|| format!("opening sqlite: {}", db_path.display()))?;

    run_migrations(&pool).await?;
    Ok(pool)
}

async fn run_migrations(pool: &DbPool) -> Result<()> {
    sqlx::query(
        r#"
        CREATE TABLE IF NOT EXISTS sandbox_outputs (
            handle TEXT PRIMARY KEY,
            session_id TEXT NOT NULL,
            tool_name TEXT NOT NULL,
            content TEXT NOT NULL,
            content_type TEXT NOT NULL,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        );
        CREATE INDEX IF NOT EXISTS idx_sandbox_session ON sandbox_outputs(session_id);

        CREATE TABLE IF NOT EXISTS journal_entries (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            session_id TEXT NOT NULL,
            tool TEXT NOT NULL,
            file_path TEXT NOT NULL,
            action TEXT NOT NULL,
            before_hash TEXT,
            after_hash TEXT,
            before_snapshot TEXT NOT NULL,
            reversible INTEGER NOT NULL DEFAULT 1,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        );
        CREATE INDEX IF NOT EXISTS idx_journal_session ON journal_entries(session_id);

        CREATE TABLE IF NOT EXISTS plans (
            id TEXT PRIMARY KEY,
            session_id TEXT NOT NULL,
            content TEXT NOT NULL,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        );

        CREATE TABLE IF NOT EXISTS todos (
            id TEXT PRIMARY KEY,
            session_id TEXT NOT NULL,
            content TEXT NOT NULL,
            status TEXT NOT NULL DEFAULT 'pending',
            priority TEXT NOT NULL DEFAULT 'medium',
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        );
        CREATE INDEX IF NOT EXISTS idx_todos_session ON todos(session_id);

        CREATE TABLE IF NOT EXISTS tasks (
            id TEXT PRIMARY KEY,
            session_id TEXT NOT NULL,
            tool_name TEXT NOT NULL,
            status TEXT NOT NULL DEFAULT 'running',
            result TEXT,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            completed_at DATETIME
        );
        CREATE INDEX IF NOT EXISTS idx_tasks_session ON tasks(session_id);
        "#,
    )
    .execute(pool)
    .await?;

    Ok(())
}

use crate::types::Message;
use anyhow::{Context, Result};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SessionMeta {
    pub session_id: String,
    pub project_path: PathBuf,
    pub title: String,
    pub created_at: chrono::DateTime<chrono::Utc>,
    pub model: String,
}

pub struct JsonlSession {
    path: PathBuf,
    meta: SessionMeta,
}

impl JsonlSession {
    pub fn create(project: &Path, title: impl Into<String>, model: impl Into<String>) -> Result<Self> {
        let session_id = format!(
            "{}-{}",
            chrono::Utc::now().format("%Y%m%d-%H%M%S"),
            uuid::Uuid::new_v4().simple().to_string().chars().take(8).collect::<String>()
        );

        let sessions_dir = crate::config::global_config_dir()
            .join("sessions")
            .join(project_hash(project));
        std::fs::create_dir_all(&sessions_dir)?;

        let path = sessions_dir.join(format!("{}.jsonl", session_id));
        let meta_path = sessions_dir.join(format!("{}.meta.json", session_id));

        let meta = SessionMeta {
            session_id: session_id.clone(),
            project_path: project.to_path_buf(),
            title: title.into(),
            created_at: chrono::Utc::now(),
            model: model.into(),
        };

        std::fs::write(&meta_path, serde_json::to_string_pretty(&meta)?)?;
        std::fs::File::create(&path)?;

        Ok(Self { path, meta })
    }

    pub fn load(session_id: &str, project: &Path) -> Result<Self> {
        let sessions_dir = crate::config::global_config_dir()
            .join("sessions")
            .join(project_hash(project));
        let path = sessions_dir.join(format!("{}.jsonl", session_id));
        let meta_path = sessions_dir.join(format!("{}.meta.json", session_id));
        let meta: SessionMeta = serde_json::from_str(&std::fs::read_to_string(meta_path)?)?;
        Ok(Self { path, meta })
    }

    pub fn list(project: Option<&Path>) -> Result<Vec<SessionMeta>> {
        let base = crate::config::global_config_dir().join("sessions");
        let mut metas = Vec::new();

        if !base.exists() {
            return Ok(metas);
        }

        for entry in std::fs::read_dir(&base)? {
            let entry = entry?;
            if let Some(proj) = project {
                if entry.file_name().to_string_lossy() != project_hash(proj) {
                    continue;
                }
            }
            for meta_entry in std::fs::read_dir(entry.path())? {
                let meta_entry = meta_entry?;
                let name = meta_entry.file_name().to_string_lossy().to_string();
                if name.ends_with(".meta.json") {
                    if let Ok(content) = std::fs::read_to_string(meta_entry.path()) {
                        if let Ok(meta) = serde_json::from_str::<SessionMeta>(&content) {
                            metas.push(meta);
                        }
                    }
                }
            }
        }

        metas.sort_by(|a, b| b.created_at.cmp(&a.created_at));
        Ok(metas)
    }

    pub fn id(&self) -> &str { &self.meta.session_id }
    pub fn meta(&self) -> &SessionMeta { &self.meta }

    pub fn append(&self, msg: &Message) -> Result<()> {
        let line = serde_json::to_string(msg)?;
        let mut file = std::fs::OpenOptions::new()
            .append(true)
            .open(&self.path)
            .with_context(|| format!("appending to {}", self.path.display()))?;
        use std::io::Write;
        writeln!(file, "{}", line)?;
        Ok(())
    }

    pub fn read_all(&self) -> Result<Vec<Message>> {
        let content = std::fs::read_to_string(&self.path)?;
        let mut msgs = Vec::new();
        for line in content.lines() {
            if line.trim().is_empty() { continue; }
            msgs.push(serde_json::from_str(line)?);
        }
        Ok(msgs)
    }

    pub fn read_tail(&self, count: usize) -> Result<Vec<Message>> {
        let all = self.read_all()?;
        let start = all.len().saturating_sub(count);
        Ok(all.into_iter().skip(start).collect())
    }
}

fn project_hash(project: &Path) -> String {
    use sha2::{Digest, Sha256};
    let mut hasher = Sha256::new();
    hasher.update(project.to_string_lossy().as_bytes());
    let result = hasher.finalize();
    result.iter().take(16).map(|b| format!("{:02x}", b)).collect()
}

use std::sync::Mutex;

pub struct FileParseResult {
    pub lines: Vec<LineInfo>,
}

pub struct LineInfo {
    pub line_number: usize,
    pub hash: String,
    pub content: String,
}

pub struct Parser {
    cache: Mutex<HashMap<PathBuf, CacheEntry>>,
}

struct CacheEntry {
    mtime: std::time::SystemTime,
    result: FileParseResult,
}

impl Parser {
    pub fn new() -> Self {
        Self {
            cache: Mutex::new(HashMap::new()),
        }
    }

    pub fn parse_file(&self, path: &Path) -> Result<FileParseResult> {
        let content = std::fs::read_to_string(path)
            .with_context(|| format!("reading file: {}", path.display()))?;
        let mtime = std::fs::metadata(path)?.modified()?;

        // Check cache
        let mut cache = self.cache.lock().unwrap();
        if let Some(entry) = cache.get(path) {
            if entry.mtime == mtime {
                return Ok(entry.result.clone());
            }
        }

        let lang = Language::from_path(path);
        let result = parse_content(&content, lang);

        cache.insert(path.to_path_buf(), CacheEntry { mtime, result: result.clone() });
        Ok(result)
    }

    pub fn get_line_hashes(&self, path: &Path) -> Result<Vec<(usize, String)>> {
        let result = self.parse_file(path)?;
        Ok(result.lines.iter().map(|l| (l.line_number, l.hash.clone())).collect())
    }

    pub fn find_hash(&self, path: &Path, hash: &str) -> Result<Option<usize>> {
        let result = self.parse_file(path)?;
        Ok(result.lines.iter().find(|l| l.hash == hash).map(|l| l.line_number))
    }

    pub fn hash_range(&self, path: &Path, start_hash: &str, end_hash: &str) -> Result<Option<(usize, usize)>> {
        let result = self.parse_file(path)?;
        let start = result.lines.iter().position(|l| l.hash == start_hash);
        let end = result.lines.iter().position(|l| l.hash == end_hash);
        match (start, end) {
            (Some(s), Some(e)) => Ok(Some((result.lines[s].line_number, result.lines[e].line_number))),
            _ => Ok(None),
        }
    }
}

fn parse_content(content: &str, _lang: Language) -> FileParseResult {
    // For M0 we just hash lines. AST-based analysis comes in M2.
    let lines = content.lines().enumerate().map(|(i, line)| {
        LineInfo {
            line_number: i + 1,
            hash: hash_line(line),
            content: line.to_string(),
        }
    }).collect();

    FileParseResult { lines }
}

pub fn hash_line(line: &str) -> String {
    let mut hasher = Sha256::new();
    hasher.update(line.as_bytes());
    let result = hasher.finalize();
    result.iter().take(8).map(|b| format!("{:02x}", b)).collect()
}

pub fn format_with_hashes(lines: &[LineInfo], start: usize, end: usize) -> String {
    lines.iter()
        .filter(|l| l.line_number >= start && l.line_number <= end)
        .map(|l| format!("  {}│{:>3}│ {}", l.hash, l.line_number, l.content))
        .collect::<Vec<_>>()
        .join("\n")
}

        let (tool_tx, tool_rx) = mpsc::channel(100);

            let mut usage = Usage::default();

                    }

            if let Some(builder) = current_tool.take() {
                if let Ok(tc) = builder.build() {
                    let _ = tool_tx.send(tc).await;
                }
            }

            let _ = usage_tx.send(usage);
        });

fn build_request(messages: &[Message], tools: &[ToolSchema], config: &crate::types::ModelConfig) -> Result<Value> {
    let msgs: Vec<Value> = messages.iter().map(to_openai_message).collect::<Result<_>>()?;

fn to_openai_message(msg: &Message) -> Result<Value> {
    match msg.role {
        Role::System => Ok(serde_json::json!({
            "role": "system",
            "content": extract_text(&msg.content),
        })),
        Role::User => Ok(serde_json::json!({
            "role": "user",
            "content": extract_text(&msg.content),
        })),
        Role::Assistant => {
            let mut content = String::new();
            let mut tool_calls = Vec::new();
            for block in &msg.content {
                match block {
                    ContentBlock::Text { text } => content.push_str(text),
                    ContentBlock::ToolUse { id, name, args } => {
                        tool_calls.push(serde_json::json!({
                            "id": id,
                            "type": "function",
                            "function": {"name": name, "arguments": args.to_string()}
                        }));
                    }
                    _ => {}
                }
            }
            if tool_calls.is_empty() {
                Ok(serde_json::json!({"role": "assistant", "content": content}))
            } else {
                Ok(serde_json::json!({"role": "assistant", "content": content, "tool_calls": tool_calls}))
            }
        }
        Role::Tool => {
            let mut result = String::new();
            let mut tool_id = String::new();
            for block in &msg.content {
                if let ContentBlock::ToolResult { id, output } = block {
                    tool_id = id.clone();
                    result = serde_json::to_string(output).unwrap_or_default();
                }
            }
            Ok(serde_json::json!({
                "role": "tool",
                "tool_call_id": tool_id,
                "content": result,
            }))
        }
    }
}

    blocks.iter().filter_map(|b| {

    arguments: Option<String>,

    completion_tokens: u32,
    total_tokens: u32,

struct ToolCallBuilder {
    id: String,
    name: String,
    args: String,
}

impl ToolCallBuilder {
    fn build(self) -> Result<ToolCall> {
        let args: Value = if self.args.is_empty() {
            serde_json::json!({})
        } else {
            serde_json::from_str(&self.args).unwrap_or(serde_json::json!({"raw": self.args}))
        };
        Ok(ToolCall { id: self.id, name: self.name, args })
    }

async fn process_delta(
    delta: SseDelta,
    text_tx: &mpsc::Sender<String>,
    tool_tx: &mpsc::Sender<ToolCall>,
    current_tool: &mut Option<ToolCallBuilder>,
    usage: &mut Usage,
) {
    if let Some(u) = delta.usage {
        usage.prompt_tokens = u.prompt_tokens;
        usage.completion_tokens = u.completion_tokens;
        usage.total_tokens = u.total_tokens;
    }

    for choice in delta.choices {
        if let Some(text) = choice.delta.content {
            let _ = text_tx.send(text).await;
        }
        if let Some(tcs) = choice.delta.tool_calls {
            for tc in tcs {
                if tc.index == 0 {
                    if let Some(id) = tc.id {
                        if current_tool.is_none() {
                            *current_tool = Some(ToolCallBuilder {
                                id,
                                name: tc.function.as_ref().and_then(|f| f.name.clone()).unwrap_or_default(),
                                args: String::new(),
                            });
                        }
                    }
                    if let Some(ref mut builder) = current_tool {
                        if let Some(f) = tc.function {
                            if let Some(name) = f.name {
                                builder.name = name;
                            }
                            if let Some(args) = f.arguments {
                                builder.args.push_str(&args);
                            }
                        }
                    }
                } else {
                    // Multi tool call: flush current, start new
                    if let Some(builder) = current_tool.take() {
                        if let Ok(tc) = builder.build() {
                            let _ = tool_tx.send(tc).await;
                        }
                    }
                    *current_tool = Some(ToolCallBuilder {
                        id: tc.id.unwrap_or_default(),
                        name: tc.function.as_ref().and_then(|f| f.name.clone()).unwrap_or_default(),
                        args: tc.function.as_ref().and_then(|f| f.arguments.clone()).unwrap_or_default(),
                    });
                }
            }
        }
    }
}

        ContentBlock::ToolUse { id, name, args } => {

async fn process_event(
    event: SseEvent,
    text_tx: &mpsc::Sender<String>,
    tool_tx: &mpsc::Sender<ToolCall>,
    tool_builders: &mut std::collections::HashMap<String, ToolCallBuilder>,
    usage: &mut Usage,
) {
    match event.event_type.as_str() {
        "message_start" => {
            if let Some(m) = event.message {
                if let Some(u) = m.usage {
                    usage.prompt_tokens = u.input_tokens;
                }
            }
        }
        "content_block_start" => {
            if let Some(cb) = event.content_block {
                if cb._type == "tool_use" {
                    let id = cb.id.unwrap_or_default();
                    tool_builders.insert(id.clone(), ToolCallBuilder {
                        id,
                        name: cb.name.unwrap_or_default(),
                        args: String::new(),
                    });
                }
            }
        }
        "content_block_delta" => {
            if let Some(delta) = event.delta {
                match delta._type.as_str() {
                    "text_delta" => {
                        if let Some(text) = delta.text {
                            let _ = text_tx.send(text).await;
                        }
                    }
                    "input_json_delta" => {
                        if let Some(json) = delta.partial_json {
                            let idx = event.index.unwrap_or(0);
                            // find corresponding tool builder (we use sequential)
                            let keys: Vec<String> = tool_builders.keys().cloned().collect();
                            if let Some(key) = keys.get(idx) {
                                if let Some(builder) = tool_builders.get_mut(key) {
                                    builder.args.push_str(&json);
                                }
                            }
                        }
                    }
                    _ => {}
                }
            }
        }
        "content_block_stop" => {}
        "message_delta" => {
            if let Some(u) = event.usage {
                usage.completion_tokens = u.output_tokens;
                usage.total_tokens = usage.prompt_tokens + usage.completion_tokens;
            }
        }
        "message_stop" => {}
        _ => {}
    }
}

#[async_trait]
impl LlmAdapter for GeminiAdapter {
    fn name(&self) -> &str { "gemini" }
    fn supports_tool_cache(&self) -> bool { true }

    async fn chat(
        &self,
        messages: &[Message],
        tools: &[ToolSchema],
        config: &crate::types::ModelConfig,
    ) -> Result<ChatStream> {
        let client = reqwest::Client::new();
        let body = build_request(messages, tools)?;

        let url = format!(
            "{}/models/{}:streamGenerateContent?key={}",
            config.base_url, config.name, config.api_key
        );

        let mut resp = client
            .post(&url)
            .json(&body)
            .send()
            .await
            .context("Gemini API request")?;

        if !resp.status().is_success() {
            let status = resp.status();
            let text = resp.text().await.unwrap_or_default();
            return Err(anyhow!("Gemini API error {}: {}", status, text));
        }

        let (text_tx, text_rx) = mpsc::channel(100);
        let (tool_tx, tool_rx) = mpsc::channel(100);
        let (usage_tx, usage_rx) = oneshot::channel();

        tokio::spawn(async move {
            let mut usage = Usage::default();
            let mut tool_builders: Vec<ToolCallBuilder> = Vec::new();

            // Gemini streaming returns JSON array of chunks, not SSE
            let mut buffer = String::new();
            while let Some(chunk) = resp.chunk().await.ok().flatten() {
                buffer.push_str(&String::from_utf8_lossy(&chunk));
                // Try to parse complete JSON arrays from buffer
                while let Some(idx) = buffer.find(']') {
                    let candidate: String = buffer.drain(..=idx).collect();
                    if let Ok(chunks) = serde_json::from_str::<Vec<GeminiChunk>>(&candidate) {
                        for chunk in chunks {
                            process_chunk(chunk, &text_tx, &tool_tx, &mut tool_builders, &mut usage).await;
                        }
                    }
                }
            }

            // Flush
            for builder in tool_builders {
                if let Ok(tc) = builder.build() {
                    let _ = tool_tx.send(tc).await;
                }
            }

            let _ = usage_tx.send(usage);
        });

        Ok(ChatStream { text_rx, tool_call_rx: tool_rx, usage: usage_rx })
    }
}

fn to_gemini_message(msg: &Message) -> Result<Value> {
    let parts: Vec<Value> = msg.content.iter().map(|block| match block {
        ContentBlock::Text { text } => {
            serde_json::json!({"text": text})
        }
        ContentBlock::ToolUse { name, args, .. } => {
            serde_json::json!({
                "functionCall": {
                    "name": name,
                    "args": args,
                }
            })
        }
        ContentBlock::ToolResult { id, output, .. } => {

        Role::User => "user",
        Role::Assistant => "model",

fn is_async_cmd(cmd: &str, timeout: u64) -> bool {
    let watch_keywords = ["watch", "&", "tail -f", "dev", "serve", "start", "hot", "reload"];

async fn run_bash_raw(
    cmd: &str,
    cwd: &str,
    timeout: u64,
    env: &std::collections::HashMap<String, String>,
    session_id: &str,
    pool: &crate::persistence::DbPool,
) -> Result<BashResult> {
    let mut command = Command::new("bash");
    command.arg("-c").arg(cmd);
    command.current_dir(cwd);
    for (k, v) in env {
        command.env(k, v);
    }
    command.stdout(Stdio::piped()).stderr(Stdio::piped());

    let mut child = command.spawn().context("spawning bash")?;
    let stdout = child.stdout.take().unwrap();
    let stderr = child.stderr.take().unwrap();

    let stdout_handle = tokio::spawn(async move {
        let reader = BufReader::new(stdout);
        let mut lines = reader.lines();
        let mut collected = Vec::new();
        while let Some(line) = lines.next_line().await.unwrap_or(None) {
            collected.push(line);
        }
        collected
    });

    let stderr_handle = tokio::spawn(async move {
        let reader = BufReader::new(stderr);
        let mut lines = reader.lines();
        let mut collected = Vec::new();
        while let Some(line) = lines.next_line().await.unwrap_or(None) {
            collected.push(line);
        }
        collected
    });

    let output = tokio::time::timeout(
        std::time::Duration::from_secs(timeout),
        child.wait(),
    ).await;

    let exit_code = match output {
        Ok(Ok(status)) => status.code().unwrap_or(-1),
        Ok(Err(e)) => return Err(anyhow::anyhow!("bash error: {}", e)),
        Err(_) => {
            // Timeout: kill the process
            let _ = child.kill().await;
            -9
        }
    };

    let stdout_lines = stdout_handle.await.unwrap_or_default();
    let stderr_lines = stderr_handle.await.unwrap_or_default();

    let stdout_full = stdout_lines.join("\n");
    let stderr_full = stderr_lines.join("\n");

    let truncated = stdout_lines.len() > 50 || stderr_lines.len() > 20;
    let stdout_summary: Vec<String> = stdout_lines.iter().rev().take(20).rev().cloned().collect();
    let stderr_summary: Vec<String> = stderr_lines.iter().rev().take(10).rev().cloned().collect();

    let full_content = format!("STDOUT:\n{}\n\nSTDERR:\n{}", stdout_full, stderr_full);
    let handle = sandbox::store_output(pool, session_id, "bash", &full_content, "text").await?;

    Ok(BashResult {
        exit_code,
        stdout_summary: stdout_summary.join("\n"),
        stderr_summary: stderr_summary.join("\n"),
        stdout_lines: stdout_lines.len(),
        stderr_lines: stderr_lines.len(),
        handle,
        truncated,
    })
}

async fn spawn_async_task(
    args: BashArgs,
    cwd: std::path::PathBuf,
    ctx: &ToolContext,
) -> Result<String> {
    let task_id = format!("t_{}", uuid::Uuid::new_v4().simple().to_string().chars().take(8).collect::<String>());
    let cmd = args.cmd;
    let timeout = args.timeout;
    let env = args.env;
    let session_id = ctx.session_id.clone();
    let pool = ctx.db_pool.clone();
    let tid = task_id.clone();

    tokio::spawn(async move {
        let result = run_bash_raw(&cmd, &cwd.to_string_lossy(), timeout, &env, &session_id, &pool).await;
        let _ = sqlx::query(
            "UPDATE tasks SET status = ?, result = ?, completed_at = CURRENT_TIMESTAMP WHERE id = ?"
        )
        .bind(match &result { Ok(_) => "completed", Err(_) => "failed" })
        .bind(serde_json::to_string(&result).ok().unwrap_or_default())
        .bind(&tid)
        .execute(&pool)
        .await;
    });

    // Create task record
    sqlx::query(
        "INSERT INTO tasks (id, session_id, tool_name, status) VALUES (?, ?, 'bash', 'running')"
    )
    .bind(&task_id)
    .bind(&ctx.session_id)
    .execute(&ctx.db_pool)
    .await?;

    Ok(task_id)
}
